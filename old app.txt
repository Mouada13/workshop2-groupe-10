import os
import sqlite3
import hashlib
from flask import Flask, request, make_response

app = Flask(__name__)

# 1. MAUVAISE PRATIQUE : Secret / Mot de passe en dur (Hardcoded Secret)
DB_PASSWORD = "SuperSecretAdminPassword123!"

# 2. INJECTION SQL (CWE-89) : Concaténation directe de l'entrée utilisateur
@app.route("/user")
def get_user():
    username = request.args.get("username", "")
    conn = sqlite3.connect("users.db")
    cursor = conn.cursor()
    query = "SELECT * FROM users WHERE username = '" + username + "'"
    cursor.execute(query)
    return str(cursor.fetchall())

# 3. INJECTION DE COMMANDE OS : Exécution directe via os.system
@app.route("/ping")
def ping():
    ip = request.args.get("ip", "127.0.0.1")
    os.system("ping -c 1 " + ip)
    return "Ping executed"

# 4. CROSS-SITE SCRIPTING - XSS (CWE-79) : Réponse HTML non échappée
@app.route("/hello")
def hello():
    name = request.args.get("name", "Guest")
    return make_response("<h1>Hello " + name + "</h1>")

# 5. MAUVAISES PRATIQUES : Utilisation de eval() et de MD5 (cryptographie faible)
@app.route("/calc")
def calc():
    expr = request.args.get("expr", "1+1")
    result = eval(expr)
    weak_hash = hashlib.md5(b"sensitive_data").hexdigest()
    return f"Result: {result}, Hash: {weak_hash}"

if __name__ == "__main__":
    app.run(debug=True)
